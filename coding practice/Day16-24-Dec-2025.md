# Day 16 — Ansible Roles, DynamoDB, StatefulSet, Service Check

## Ansible — Create Role Structure

1. roles/pathnex/vars/main.yml
---
app_name: pathnex
http_port: 80
web_root: /var/www/html
package_name: httpd
service_name: httpd

2. roles/pathnex/tasks/main.yml
---
- name: Install web server package
  yum:
    name: "{{ package_name }}"
    state: present

- name: Ensure web service is started
  service:
    name: "{{ service_name }}"
    state: started
    enabled: yes

- name: Deploy application index page
  template:
    src: index.html.j2
    dest: "{{ web_root }}/index.html"
  notify: Restart web service


3. roles/pathnex/handlers/main.yml

---
- name: Restart web service
  service:
    name: "{{ service_name }}"
    state: restarted

4. roles/pathnex/templates/index.html.j2
<!DOCTYPE html>
<html>
<head>
  <title>{{ app_name }} App</title>
</head>
<body>
  <h1>Welcome to {{ app_name }}</h1>
  <p>Deployed using Ansible Role</p>
</body>
</html>

##  Shell Script — Check Service Running

#!/bin/bash

SERVICE="sshd"

if systemctl is-active --quiet $SERVICE; then
  echo "$SERVICE is running"
else
  echo "$SERVICE is NOT running"
fi


# Artifacts Promotion
##  Jenkins Pipeline — Promote Artifacts

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
        stage('Build') {
            steps {
                sh 'mvn clean package'
            }
        }
        stage('Promote') {
            steps {
                input message: "Promote artifacts to staging?"
                sh 'echo "Promoting artifacts for $INSTITUTE_NAME"'
            }
        }
    }
}

##  GitLab CI — Artifacts Promotion

stages:
  - build
  - promote

build:
  stage: build
  image: maven:3.8.1-jdk-17
  script:
    - mvn clean package
  artifacts:
    paths:
      - target/*.jar
    expire_in: 1 week

promote:
  stage: promote
  image: alpine:latest
  script:
    - echo "Promoting artifacts for Pathnex"
  when: manual
