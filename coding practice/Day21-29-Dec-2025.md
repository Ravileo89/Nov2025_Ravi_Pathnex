##  Jenkins Pipeline — Deploy to Multiple Environments

pipeline {
  agent any
  environment {
      INSTITUTE_NAME = "Pathnex"
  }
  stages {
      stage('Deploy to Dev') {
          steps {
              sh 'echo "Deploying $INSTITUTE_NAME to the dev environment"'
          }
      }
      stage('Deploy to Stage') {
          steps {
              sh 'echo "Deploying $INSTITUTE_NAME to the Staging environment"'
          }
      }
      stage('Deploy to the Prod') {
          steps {
              input message: "approve to production environment "
              sh 'echo "Deploying $INSTITUTE_NAME to the Production environment"'
          }
      }
  }
}
